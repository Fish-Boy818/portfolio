        OrderResponse order = orderService.createOrder(request);
        return ApiResponse.ok(order);
    }

        return ApiResponse.ok(orderService.getOrderDetail(id));
    }

@Service
public class OrderService {
    @Transactional
    public OrderResponse markPaid(Long orderId, PayNotify notify) {
        Order order = orderMapper.selectById(orderId);
        validatePayNotify(order, notify);

        order.setStatus(OrderStatus.PAID);
        order.setPaidAt(LocalDateTime.now());
        orderMapper.update(order);

        commissionService.createPendingRecords(order);
        return OrderResponse.from(order);
    }

PyTorch：字符级编码与情感推理
def encode_text(text: str, vocab: dict[str, int], max_len: int) -> list[int]:
    ids = [vocab.get(token, UNK_ID) for token in tokenize(text)]
    ids = ids[:max_len]
    return ids + [PAD_ID] * (max_len - len(ids))


def predict(text: str, model, checkpoint: dict, device: torch.device) -> dict:
    input_ids = encode_text(text, checkpoint["vocab"], checkpoint["max_len"])
    tensor = torch.tensor([input_ids], dtype=torch.long, device=device)
    with torch.no_grad():
        probabilities = torch.softmax(model(tensor), dim=1).squeeze(0)
    class_id = int(probabilities.argmax().item())
    return {"text": text, "label": checkpoint["class_names"][class_id]}
